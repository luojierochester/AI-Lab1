cmake_minimum_required(VERSION 3.16)
project(NPuzzleVisualizer VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(NPZ_BUILD_SFML "Build the pinned SFML source distribution" ON)
if(NPZ_BUILD_SFML)
    include(FetchContent)
    set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_DOC OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(SFML_USE_STATIC_STD_LIBS ON CACHE BOOL "" FORCE)
    FetchContent_Declare(sfml
        URL https://www.sfml-dev.org/files/SFML-2.6.2-sources.zip
        URL_HASH SHA256=19d6dbd9c901c74441d9888c13cb1399f614fe8993d59062a72cfbceb00fed04
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(sfml)
else()
    find_package(SFML 2.5 COMPONENTS graphics window system REQUIRED)
endif()

add_executable(npuzzle-viewer src/main.cpp)
target_link_libraries(npuzzle-viewer PRIVATE sfml-graphics sfml-window sfml-system)
if(MINGW)
    target_compile_options(npuzzle-viewer PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(npuzzle-viewer PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(WIN32)
    target_link_libraries(npuzzle-viewer PRIVATE comdlg32 shell32 imm32)
endif()
enable_testing()
add_test(NAME viewer-checks COMMAND npuzzle-viewer --self-test)
